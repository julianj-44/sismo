# Visual de R de Power BI: Colombia en detalle (ggplot2, imagen estática)
# Campos en Valores (todos con "No resumir"): latitud, longitud, grupo, magnitud, fecha_evento, profundidad_km
# Paquetes: install.packages(c("ggplot2", "maps"))
#
# Cómo se lee el mapa:
#   tamaño = magnitud
#   color  = profundidad (más oscuro = más profundo)
#   anillo = evento de las últimas 72 h respecto al evento más reciente del conjunto

# Pegue o escriba aquí el código de script:

library(ggplot2)
library(maps)

# --- Fechas (hora UTC sin zona; se trata como UTC) -----------------------------------------
if (!"fecha_evento" %in% names(dataset)) dataset$fecha_evento <- NA
fe <- dataset$fecha_evento   # fecha/hora (Power BI) o texto (editor externo)
dataset$fecha <- if (inherits(fe, c("POSIXt", "Date"))) {
  as.POSIXct(format(fe, "%Y-%m-%d %H:%M:%S"), tz = "UTC")
} else as.POSIXct(as.character(fe), tz = "UTC")
ref <- if (all(is.na(dataset$fecha))) NA else max(dataset$fecha, na.rm = TRUE)

# Solo Colombia y sin magnitud faltante (no se reemplaza por 0: se omite del mapa)
dataset <- dataset[dataset$grupo == "Colombia" & !is.na(dataset$magnitud), ]
tiene_prof <- "profundidad_km" %in% names(dataset) && any(!is.na(dataset$profundidad_km))

dataset$reciente <- if (is.na(ref)) FALSE else
  !is.na(dataset$fecha) & as.numeric(difftime(ref, dataset$fecha, units = "hours")) <= 72
recientes <- dataset[dataset$reciente, ]

# --- Contexto: Colombia resaltada y vecinos en gris ----------------------------------------
vecinos  <- map_data("world", region = c("Colombia", "Venezuela", "Ecuador", "Peru", "Brazil",
                                         "Panama", "Costa Rica", "Nicaragua"))
colombia <- vecinos[vecinos$region == "Colombia", ]

# Nota: linewidth necesita ggplot2 3.4 o superior; en versiones anteriores usar size.
g <- ggplot() +
  geom_polygon(data = vecinos,  aes(long, lat, group = group),
               fill = "#E3EAF4", colour = "#B7C4D8", linewidth = 0.25) +
  geom_polygon(data = colombia, aes(long, lat, group = group),
               fill = "#FFE3CC", colour = "#FF7A1A", linewidth = 0.6)

if (tiene_prof) {
  g <- g +
    geom_point(data = dataset, aes(x = longitud, y = latitud, size = magnitud, fill = profundidad_km),
               shape = 21, colour = "#07142B", stroke = 0.5, alpha = 0.92) +
    scale_fill_gradient(low = "#FFC999", high = "#7A2E00", name = "Profundidad (km)",
                        guide = guide_colourbar(order = 1, barwidth = unit(90, "pt"), barheight = unit(7, "pt")))
} else {
  g <- g +
    geom_point(data = dataset, aes(x = longitud, y = latitud, size = magnitud),
               shape = 21, fill = "#FF7A1A", colour = "#07142B", stroke = 0.5, alpha = 0.92)
}

g +
  geom_point(data = recientes, aes(x = longitud, y = latitud, size = magnitud),
             shape = 21, fill = NA, colour = "#0B1B33", stroke = 0.9, show.legend = FALSE) +
  scale_size_continuous(range = c(2.5, 8), name = "Magnitud",
                        guide = guide_legend(order = 2, override.aes = list(fill = "#94A3B8"))) +
  coord_quickmap(xlim = c(-79.5, -66.5), ylim = c(-4.5, 13.5), expand = FALSE) +
  labs(caption = paste0(nrow(dataset), " eventos en Colombia · anillo: últimos 72 h")) +
  theme_void() +
  theme(legend.position = "bottom", legend.box = "vertical", legend.box.just = "left",
        legend.margin = margin(t = 4, b = 2),
        legend.text  = element_text(colour = "#475569", size = 9),
        legend.title = element_text(colour = "#475569", size = 9, face = "bold"),
        plot.caption = element_text(colour = "#64748B", size = 9, hjust = 0, margin = margin(t = 2)),
        panel.background = element_rect(fill = "#EAF0F8", colour = NA),
        plot.background  = element_rect(fill = "white", colour = NA))
